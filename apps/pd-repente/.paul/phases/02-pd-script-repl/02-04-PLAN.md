## Goal
Add `/help <topic>` paged help and a synchronous `pds` Lua table
(pds.create/connect/delete/move/list) so Lua loops can create and wire
objects in a single expression block.

## Purpose
Completes Phase 02: the REPL is now scriptable. Generative patching
(`for i=1,8 do pds.create("osc~", i*60, 100) end`) works end-to-end.
LLM-generated Lua becomes runnable when Phase 03 ships.

- PromptInput: `/help` with no args → compact overview; `/help pds|sugar|lua|llm|commands` → topic detail
- Executor: `executeSync(cmd)` — synchronous execute, returns result string; safe on message thread
- CommandInput.h: `LuaExpressionParser::registerExtension(fn)` + `CommandInput::registerLuaExtension(fn)` protected hook
- PromptInput: `pds` global Lua table registered on construction; pds.create returns assigned name
</objective>

    ```cpp
    // Lua C closures — upvalue 1 = PromptInput*
    static int lua_pds_create(lua_State* L)
    {
        auto* self = static_cast<PromptInput*>(lua_touserdata(L, lua_upvalueindex(1)));
        juce::String objText = luaL_checkstring(L, 1);
        int x = (int)luaL_optinteger(L, 2, 100);
        int y = (int)luaL_optinteger(L, 3, 100);
        auto cmd = CommandParser::parse(
            "/pds create " + objText + " " + juce::String(x) + " " + juce::String(y));
        auto result = self->executor->executeSync(cmd);
        if (self->pluginEditor)
            if (auto* pd = self->pluginEditor->pd)
                pd->logMessage(result);
        if (self->onRegistryChanged) self->onRegistryChanged();
        lua_pushstring(L, self->executor->getLastCreatedName().toRawUTF8());
        return 1;
    }

    static int lua_pds_connect(lua_State* L)
    {
        auto* self = static_cast<PromptInput*>(lua_touserdata(L, lua_upvalueindex(1)));
        juce::String a   = luaL_checkstring(L, 1);
        int nout         = (int)luaL_optinteger(L, 2, 0);
        juce::String b   = luaL_checkstring(L, 3);
        int nin          = (int)luaL_optinteger(L, 4, 0);
        auto cmd = CommandParser::parse(
            "/pds connect " + a + " " + juce::String(nout) + " " + b + " " + juce::String(nin));
        auto result = self->executor->executeSync(cmd);
        if (self->pluginEditor)
            if (auto* pd = self->pluginEditor->pd)
                pd->logMessage(result);
        return 0;
    }

    static int lua_pds_delete(lua_State* L)
    {
        auto* self = static_cast<PromptInput*>(lua_touserdata(L, lua_upvalueindex(1)));
        juce::String name = luaL_checkstring(L, 1);
        auto cmd = CommandParser::parse("/pds delete " + name);
        auto result = self->executor->executeSync(cmd);
        if (self->pluginEditor)
            if (auto* pd = self->pluginEditor->pd)
                pd->logMessage(result);
        if (self->onRegistryChanged) self->onRegistryChanged();
        return 0;
    }

    static int lua_pds_move(lua_State* L)
    {
        auto* self = static_cast<PromptInput*>(lua_touserdata(L, lua_upvalueindex(1)));
        juce::String name = luaL_checkstring(L, 1);
        int x = (int)luaL_checkinteger(L, 2);
        int y = (int)luaL_checkinteger(L, 3);
        auto cmd = CommandParser::parse(
            "/pds move " + name + " " + juce::String(x) + " " + juce::String(y));
        auto result = self->executor->executeSync(cmd);
        if (self->pluginEditor)
            if (auto* pd = self->pluginEditor->pd)
                pd->logMessage(result);
        return 0;
    }

    static int lua_pds_list(lua_State* L)
    {
        auto* self = static_cast<PromptInput*>(lua_touserdata(L, lua_upvalueindex(1)));
        auto cmd = CommandParser::parse("/pds list");
        auto result = self->executor->executeSync(cmd);
        if (self->pluginEditor)
            if (auto* pd = self->pluginEditor->pd)
                pd->logMessage(result);
        return 0;
    }

    // Called once from constructor via registerLuaExtension
    void PromptInput::registerPdsTable(lua_State* L)
    {
        lua_newtable(L);

        struct { const char* name; lua_CFunction fn; } fns[] = {
            { "create",  lua_pds_create  },
            { "connect", lua_pds_connect },
            { "delete",  lua_pds_delete  },
            { "move",    lua_pds_move    },
            { "list",    lua_pds_list    },
        };
        for (auto const& f : fns) {
            lua_pushlightuserdata(L, this);
            lua_pushcclosure(L, f.fn, 1);
            lua_setfield(L, -2, f.name);
        }

        lua_setglobal(L, "pds");
    }
    ```
