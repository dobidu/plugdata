    ```cpp
    #include <juce_audio_basics/juce_audio_basics.h>
    #include <juce_dsp/juce_dsp.h>
    #include <vector>

    namespace RepentePd {

    class SpectralAnalyzer {
    public:
        struct Result {
            float sub = 0.f, low = 0.f, midLow = 0.f, midHigh = 0.f, high = 0.f;
            std::vector<float> peakFrequencies; // Hz, up to 3, descending magnitude
            int sampleRate = 44100;
        };

        static Result analyze(juce::AudioBuffer<float> const& buffer, int sampleRate);
        static juce::String format(Result const& r);
    };

    } // namespace RepentePd
    ```

    **SpectralAnalyzer.cpp — analyze():**

    Constants:
    - fftOrder = 11  →  fftSize = 2048
    - hopSize = 1024 (50% overlap)
    - Buffer for FFT: `std::vector<float> fftData(fftSize * 2, 0.f)`

    Steps:
    1. **Mix to mono**: average all channels into a `std::vector<float> mono(numSamples)`
       ```cpp
       for each channel ch: for each sample i: mono[i] += buffer.getSample(ch, i) / numChannels
       ```

    2. **Build Hann window**: precompute `std::vector<float> window(fftSize)`
       ```cpp
       for (int i = 0; i < fftSize; ++i)
           window[i] = 0.5f * (1.0f - std::cos(2.0f * juce::MathConstants<float>::pi * i / (fftSize - 1)));
       ```

    3. **Multi-frame accumulation**: accumulate magnitude spectrum into `std::vector<float> avgSpectrum(fftSize, 0.f)`, frameCount = 0
       For each hop starting at offset 0, step hopSize, while offset + fftSize <= numSamples:
       - Fill fftData[0..fftSize-1] with windowed mono samples: `fftData[i] = mono[offset + i] * window[i]`
       - Zero fftData[fftSize..2*fftSize-1]
       - Create `juce::dsp::FFT fft(fftOrder)` and call `fft.performRealOnlyForwardTransform(fftData.data())`
       - Accumulate magnitude per bin: `avgSpectrum[i] += std::hypot(fftData[i*2], fftData[i*2+1])` for i in [0, fftSize/2]
       - frameCount++

    4. **Normalize by frame count**: if frameCount > 0, divide each avgSpectrum[i] by frameCount

    5. **Compute band energies**: sum avgSpectrum bins in each band
       Band boundaries (compute as bin index = round(freqHz * fftSize / sampleRate)):
       - sub:    0 Hz  – 60 Hz
       - low:    60 Hz – 250 Hz
       - midLow: 250 Hz – 2000 Hz
       - midHigh: 2000 Hz – 6000 Hz
       - high:   6000 Hz – min(20000 Hz, nyquist)
       Sum magnitude values in each bin range.

    6. **Normalize bands to [0,1]**: find max band energy; if > 0, divide all by max.

    7. **Peak detection**: find top 3 local maxima in avgSpectrum[1..fftSize/2-1]
       (exclude bin 0 = DC). Local max: `avgSpectrum[i] > avgSpectrum[i-1] && avgSpectrum[i] > avgSpectrum[i+1]`
       Collect all local maxima as (magnitude, bin) pairs, sort descending by magnitude, take top 3.
       Convert to Hz: `hz = static_cast<float>(bin) * sampleRate / fftSize`

    8. **Handle silence**: if all avgSpectrum values are 0 (or no frames), return default Result with zeros and empty peaks.

    Avoid: creating juce::dsp::FFT inside the hot loop repeatedly (it's fine for analyze() since it's called once per /listen, not real-time).
    Avoid: std::sort on the full spectrum — collect local maxima first, then sort that small list.

    **SpectralAnalyzer.cpp — format():**
    ```cpp
    juce::String SpectralAnalyzer::format(Result const& r)
    {
        auto fmt = [](float v) { return juce::String(v, 2); };
        juce::String s = "spectral: sub=" + fmt(r.sub)
            + " low=" + fmt(r.low)
            + " mid=" + fmt(r.midLow)
            + " hmid=" + fmt(r.midHigh)
            + " high=" + fmt(r.high);
        if (!r.peakFrequencies.empty()) {
            s += " | peaks:";
            for (auto hz : r.peakFrequencies)
                s += " " + juce::String(static_cast<int>(hz)) + "Hz";
        }
        return s;
    }
    ```
