<objective>
## Goal
Persist Repente bridge config (URL/model/key) in SettingsFile, expose `/config`
command for runtime configuration, and add `/config test` to ping the server.

## Purpose
Without persistent config, Bridge is hardcoded to localhost:7860 — unusable with
any real server. This plan makes the bridge actually configurable so the Phase 03
end-to-end demo is completable with a real OpenAI-compat server.

## Output
- SettingsFile keys `repente_url`, `repente_model`, `repente_key` loaded on startup
- `/config` command: show/set URL/model/key, test connection
- `RepenteClient::ping()` for connection test
- `/help llm` updated to reference `/config`
</objective>

    ### RepenteClient.cpp — implement ping()

    Add after `send()` implementation:
    ```cpp
    void RepenteClient::ping(std::function<void(bool, juce::String)> callback)
    {
        jassert(juce::MessageManager::getInstance()->isThisTheMessageThread());

        auto cfg = config; // copy before thread
        std::thread([cfg, callback]() mutable {
            httplib::Client cli(cfg.host.toStdString(), cfg.port);
            cli.set_connection_timeout(5, 0);
            cli.set_read_timeout(5, 0);

            auto path = (cfg.pathPrefix + "/models").toStdString();
            auto res = cli.Get(path);

            juce::String msg;
            bool ok = false;
            if (res) {
                ok = (res->status == 200);
                if (ok) {
                    try {
                        auto j = nlohmann::json::parse(res->body);
                        int count = (int)(j.contains("data") ? j["data"].size() : 0);
                        msg = "models available: " + juce::String(count);
                    } catch (...) {
                        msg = "connected (status 200)";
                    }
                } else {
                    msg = "HTTP " + juce::String(res->status);
                }
            } else {
                msg = juce::String(httplib::to_string(res.error()).c_str());
            }

            callAsync([callback, ok, msg]() mutable {
                callback(ok, msg);
            });
        }).detach();
    }
    ```

    Note: include `<nlohmann/json.hpp>` is already in RepenteClient.cpp.
    `callAsync`, `cfg.host`, `cfg.port`, `cfg.pathPrefix` are already used in `send()` — use same pattern.

    ### PromptInput.cpp — add /config routing

    In `executeCommand()`, add a `/config` handler BEFORE the `/pds` block
    (after the `/help` and `/clear` handlers):

    ```cpp
    if (msg.startsWith("/config")) {
        auto args = msg.substring(7).trim();

        if (args.isEmpty()) {
            // Show current config
            auto const& cfg = bridge ? bridge->getConfig()
                                     : RepentePd::RepenteClient::Config{};
            juce::String masked = cfg.apiKey.isNotEmpty() ? "****" : "(not set)";
            pdInstance->logMessage(
                "repente config:\n"
                "  url:   " + cfg.url + "\n"
                "  model: " + cfg.model + "\n"
                "  key:   " + masked);
            return {};
        }

        if (args.startsWith("url ")) {
            juce::String newUrl = args.substring(4).trim();
            SettingsFile::getInstance()->setProperty("repente_url", newUrl);
            if (bridge) {
                auto cfg = bridge->getConfig();
                cfg.url = newUrl;
                bridge->setConfig(std::move(cfg));
            }
            pdInstance->logMessage("repente: url set to " + newUrl);
            return {};
        }

        if (args.startsWith("model ")) {
            juce::String newModel = args.substring(6).trim();
            SettingsFile::getInstance()->setProperty("repente_model", newModel);
            if (bridge) {
                auto cfg = bridge->getConfig();
                cfg.model = newModel;
                bridge->setConfig(std::move(cfg));
            }
            pdInstance->logMessage("repente: model set to " + newModel);
            return {};
        }

        if (args.startsWith("key ")) {
            juce::String newKey = args.substring(4).trim();
            SettingsFile::getInstance()->setProperty("repente_key", newKey);
            if (bridge) {
                auto cfg = bridge->getConfig();
                cfg.apiKey = newKey;
                bridge->setConfig(std::move(cfg));
            }
            pdInstance->logMessage("repente: api key set (masked)");
            return {};
        }

        if (args == "test") {
            if (!bridge) {
                pdInstance->logMessage("repente: bridge not ready");
                return {};
            }
            auto url = bridge->getConfig().url;
            pdInstance->logMessage("repente: testing " + url + "...");
            bridge->ping([pd = pdInstance](bool ok, juce::String const& msg) {
                if (ok)
                    pd->logMessage("repente: connected — " + msg);
                else
                    pd->logMessage("repente: connection failed — " + msg);
            });
            return {};
        }

        pdInstance->logMessage(
            "usage:\n"
            "  /config              -- show current config\n"
            "  /config url <url>    -- set server URL\n"
            "  /config model <name> -- set model name\n"
            "  /config key <key>    -- set API key\n"
            "  /config test         -- test server connection");
        return {};
    }
    ```
