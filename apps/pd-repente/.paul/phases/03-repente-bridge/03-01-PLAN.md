    ### RepenteClient.h

    ```cpp
    #pragma once
    #include <juce_events/juce_events.h>
    #include <functional>

    namespace RepentePd {

    class RepenteClient {
    public:
        struct Config {
            juce::String url     = "http://localhost:7860";
            juce::String model   = "repente-1";
            juce::String apiKey  = "";
            int          timeoutSec = 30;
        };

        explicit RepenteClient(Config cfg = {});
        ~RepenteClient();

        // Fire-and-forget: spawns background thread, fires callback on message thread.
        // callback(result): result is response content string, or "error: <reason>".
        void send(juce::String const& prompt,
                  std::function<void(juce::String)> callback);

        void setConfig(Config cfg);
        Config const& getConfig() const { return config; }

        // Returns true if a request is in flight.
        bool isBusy() const;

    private:
        Config config;
        std::atomic<bool> busy { false };

        JUCE_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR(RepenteClient)
    };

    } // namespace RepentePd
    ```

    ### RepenteClient.cpp

    ```cpp
    #include "Utility/Config.h"
    #include "RepenteClient.h"

    // httplib and json must be included ONLY in .cpp to keep compile times sane.
    // Define CPPHTTPLIB_OPENSSL_SUPPORT before include only if OpenSSL found —
    // use a compile-definition from CMake rather than hardcoding (see note below).
    #include <cpp-httplib/httplib.h>
    #include <nlohmann/json.hpp>
    using json = nlohmann::json;

    namespace RepentePd {

    RepenteClient::RepenteClient(Config cfg) : config(std::move(cfg)) {}
    RepenteClient::~RepenteClient() = default;

    void RepenteClient::setConfig(Config cfg) { config = std::move(cfg); }
    bool RepenteClient::isBusy() const { return busy.load(); }

    void RepenteClient::send(juce::String const& prompt,
                             std::function<void(juce::String)> callback)
    {
        if (busy.exchange(true)) {
            juce::MessageManager::callAsync([cb = std::move(callback)] {
                cb("error: client busy");
            });
            return;
        }

        // Capture config by value for the thread
        auto cfg      = config;
        auto cbCopy   = std::move(callback);
        auto* busyPtr = &busy;

        std::thread([cfg, cbCopy, busyPtr]() mutable {
            juce::String result;
            try {
                // Parse base URL into host + path prefix
                juce::String urlStr = cfg.url;
                // Strip trailing slash
                if (urlStr.endsWithChar('/')) urlStr = urlStr.dropLastCharacters(1);

                // Extract host from URL (strip http:// or https://)
                juce::String host = urlStr;
                bool useHttps = host.startsWith("https://");
                host = host.removeCharacters("https://").removeCharacters("http://");
                // Safer: use proper stripping
                if (urlStr.startsWith("https://")) host = urlStr.substring(8);
                else if (urlStr.startsWith("http://")) host = urlStr.substring(7);
                else host = urlStr;

                // Split host:port/path
                int port = useHttps ? 443 : 80;
                juce::String pathPrefix;
                int colonPos = host.indexOf(":");
                int slashPos = host.indexOf("/");
                if (slashPos >= 0) {
                    pathPrefix = host.substring(slashPos);
                    host = host.substring(0, slashPos);
                }
                if (colonPos >= 0 && (slashPos < 0 || colonPos < slashPos)) {
                    port = host.substring(colonPos + 1).getIntValue();
                    host = host.substring(0, colonPos);
                }

                std::string endpoint = (pathPrefix + "/v1/chat/completions").toStdString();

                // Build OpenAI-compat body
                json body = {
                    {"model", cfg.model.toStdString()},
                    {"messages", {{{"role", "user"}, {"content", prompt.toStdString()}}}},
                    {"stream", false}
                };
                std::string bodyStr = body.dump();

                httplib::Client cli(host.toStdString(), port);
                cli.set_connection_timeout(cfg.timeoutSec);
                cli.set_read_timeout(cfg.timeoutSec);

                httplib::Headers headers = {
                    {"Content-Type", "application/json"},
                    {"Accept", "application/json"}
                };
                if (cfg.apiKey.isNotEmpty())
                    headers.emplace("Authorization", "Bearer " + cfg.apiKey.toStdString());

                auto res = cli.Post(endpoint, headers, bodyStr, "application/json");

                if (!res) {
                    result = "error: " + juce::String(httplib::to_string(res.error()).c_str());
                } else if (res->status != 200) {
                    result = "error: HTTP " + juce::String(res->status);
                } else {
                    json resp = json::parse(res->body, nullptr, false);
                    if (resp.is_discarded()) {
                        result = "error: invalid JSON response";
                    } else {
                        auto content = resp["choices"][0]["message"]["content"].get<std::string>();
                        result = juce::String(content.c_str());
                    }
                }
            }
            catch (std::exception const& e) {
                result = "error: " + juce::String(e.what());
            }

            busyPtr->store(false);
            juce::MessageManager::callAsync([cbCopy, result] {
                cbCopy(result);
            });
        }).detach();
    }

    } // namespace RepentePd
    ```
